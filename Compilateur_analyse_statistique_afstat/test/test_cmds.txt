===========================
Détection d’erreurs (négatifs, messages attendus)
===========================

# [KO LEXICAL] Caractère ou lexème non reconnu par le lexer
# -> Doit produire une erreur lexicale (avant parsing)
make test_lexical_invalid

# [KO SYNTAX] Programme ne respectant pas la grammaire (ex: ; manquant, bloc mal fermé)
# -> Doit produire une erreur de syntaxe via Bison (yyerror)
make test_syntax_error

# [KO SEMANTIC] Colonne inconnue dans un schéma/dataset (référence invalide)
# -> identifiant/colonne non déclarée
make test_semantic_unknown_column

# [KO SEMANTIC] Join sur colonnes incompatibles (types différents non compatibles)
# ->  erreur de compatibilité de jointure
make test_semantic_join_mismatch

# [KO SEMANTIC] Analyse incompatible ( mean sur colonne string)
# ->  vérification type d'opération analyze vs type de colonne
make test_analyze_type_mismatch

# [KO SEMANTIC] Expression invalide ("abc" + 3, opérandes incompatibles)
# -> vérification de types dans compute/expressions
make test_expr_type_mismatch

# [KO SEMANTIC] Identifiant non défini (dataset ou variable non déclarée)
# ->  résolution des identifiants
make test_undefined_identifier

# [KO SCHEMA/SOURCE] Schéma incohérent avec le CSV associé (type ou structure incorrecte)
# ->  mismatch entre schéma et données
make test_schema_type_mismatch

# [KO SCHEMA/SOURCE] Schéma invalide / CSV non conforme (colonnes manquantes, format inattendu)
# -> validation stricte schema/source
make test_bad_schema
